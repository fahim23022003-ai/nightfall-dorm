func machine_ready(room,kind: String,goal: int=1,number: int=1) -> bool:

func choose_room(model) -> int:
 var best=-1;var best_score=INF
 for room in model.rooms:
  if room.owner>=0:continue
  var path=model.player_navigation_path(model.player_position,room.bed_position()+Vector2(31,49))
  if path.is_empty():continue
  var score=0.0;var previous=model.player_position
  for point in path:score+=previous.distance_to(point);previous=point
  if room.trait_id=="fortified":score-=200
  if score<best_score:best=room.index;best_score=score
 return best
